Devices Sold · Pro3 + Smart2 · Canal Resellers

BT_PRODUCT_ORDERS BT_PRODUCT_ORDER_ITEM BT_MP_POINT_USERS_GROUPS BT_MP_POINT_GROUP 
-- Devices Sold por período (trocar datas para cada período)
-- Q1: rodar para Jan/Fev/Mar 01–18 e dividir soma por 3
SELECT
    oi.PI_NAME                     AS device,
    COUNT(oi.PI_ORDER_ITEM_ID)    AS qtd
FROM `meli-bi-data.WHOWNER.BT_PRODUCT_ORDERS` o
JOIN `meli-bi-data.WHOWNER.BT_PRODUCT_ORDER_ITEM` oi
    ON o.IDX_ID = oi.IDX_ID
LEFT JOIN `meli-bi-data.WHOWNER.BT_MP_POINT_USERS_GROUPS` ug
    ON o.CUS_CUST_ID_BUY = ug.CUS_CUST_ID AND ug.ACTIVE = 1
LEFT JOIN `meli-bi-data.WHOWNER.BT_MP_POINT_GROUP` gr
    ON ug.GROUP_ID = gr.ID_GROUP
   AND gr.GROUP_TYPE = 'RESELLER' AND gr.ACTIVE = 1
WHERE o.ORDER_GATEWAY  = '1'
  AND o.SIT_SITE_ID    = 'MLB'
  AND o.OPERATION_TYPE = '1'
  AND DATE(o.PO_CREATED_AT) BETWEEN '2026-06-01' AND '2026-06-18'
  AND oi.PI_NAME IN ('Point Pro 3', 'Point Smart 2')
GROUP BY 1

TPV M0 · Jun/26 · dias 1-18

BD_CUST_RESELLER_INFO Schema: SBOX_WHMPBR 
-- Jun/26: fonte RESELLER_INFO pois CUBO_RSL não fecha correto no mês corrente.
-- NIVEL NULL em junho → join com maio para classificação por nível (se necessário).
-- MONTH_ é NUMERIC no formato YYYYMM.

SELECT SUM(TPV_M0) AS tpv_m0_jun
FROM `meli-bi-data.SBOX_WHMPBR.BD_CUST_RESELLER_INFO`
WHERE MONTH_ = 202606

TPV M0 · Mai/26 · dias 1-18

FROM `meli-bi-data.WHOWNER.BT_MP_IC_CUBO_RSL`

TPV M0 · Média Q1 (Jan–Mar) · dias 1-18

BT_MP_IC_CUBO_RSL Schema: WHOWNER 
-- Soma os 3 meses e divide por 3 para obter a média mensal
SELECT SUM(TPV) / 3 AS tpv_m0_q1_media
FROM `meli-bi-data.WHOWNER.BT_MP_IC_CUBO_RSL`
WHERE (
    TIM_DAY BETWEEN '2026-01-01' AND '2026-01-18' OR
    TIM_DAY BETWEEN '2026-02-01' AND '2026-02-18' OR
    TIM_DAY BETWEEN '2026-03-01' AND '2026-03-18'
)
  AND CICLO_VIDA_M0_TRIANGULO = 'M0'
  AND PRODUTO_MACRO            = 'POINT'
