🔍 Queries Utilizadas

1. Levantamento de campos — LK_MP_IC_SELLERS_PROFILE_RSL
Usada para descobrir os campos disponíveis na tabela de perfil de resellers antes de montar a análise.
SELECT
  column_name,
  data_type
FROM `meli-bi-data.WHOWNER.INFORMATION_SCHEMA.COLUMNS`
WHERE table_name = 'LK_MP_IC_SELLERS_PROFILE_RSL'
ORDER BY ordinal_position
2. Validação de níveis — todos os resellers têm nível?
Verificação de integridade: confirma se há resellers sem NIVEL_RESELLER identificado entre os que se qualificam para a campanha.
-- Agrupa por nível (nulos viram 'SEM NIVEL') para verificar integridade
WITH sellers_m0_junho AS (
  SELECT
    NIVEL_RESELLER,
    RESELLER_ID,
    CUS_CUST_ID,
    SUM(TPV) AS tpv_junho
  FROM `meli-bi-data.WHOWNER.BT_MP_IC_CUBO_RSL`
  WHERE TIM_DAY >= '2026-06-01'
    AND TIM_DAY < '2026-07-01'
    AND MESES_DESDE_AQUISICAO = 0
  GROUP BY 1,2,3
),
filtro_tpv AS (
  SELECT * FROM sellers_m0_junho WHERE tpv_junho >= 3000
),
por_reseller AS (
  SELECT
    COALESCE(NULLIF(TRIM(NIVEL_RESELLER), ''), 'SEM NIVEL') AS nivel,
    RESELLER_ID,
    COUNT(CUS_CUST_ID)  AS novos_clientes,
    SUM(tpv_junho)      AS tpv_total
  FROM filtro_tpv
  GROUP BY 1,2
)
SELECT
  nivel,
  COUNT(DISTINCT RESELLER_ID) AS qtd_resellers,
  SUM(novos_clientes)         AS total_novos,
  SUM(tpv_total)              AS tpv_agregado
FROM por_reseller
GROUP BY 1
ORDER BY qtd_resellers DESC
3. Ranking final — Top 10 por nível com desempate por TPV
Query principal que gera o ranking. Sellers M0 de Point com TPV ≥ R$3.000. Critério primário: maior número de novos clientes. Desempate: maior TPV M0 acumulado no mês.
-- Sellers M0 de Point em junho com TPV >= 3k, rankeados por reseller
-- Desempate: maior TPV acumulado M0 vence
WITH sellers_m0 AS (
  SELECT
    NIVEL_RESELLER,
    RESELLER_ID,
    CUS_CUST_ID,
    SUM(TPV) AS tpv_junho
  FROM `meli-bi-data.WHOWNER.BT_MP_IC_CUBO_RSL`
  WHERE TIM_DAY >= '2026-06-01'
    AND TIM_DAY < '2026-07-01'
    AND CICLO_VIDA_M0_TRIANGULO = 'M0'  -- primeiro mês de transação do seller
    AND PRODUTO = 'POINT'              -- somente Point (maquininha)
  GROUP BY 1,2,3
),
filtro_tpv AS (
  -- Corte mínimo de R$ 3.000 de faturamento Point por seller no mês
  SELECT * FROM sellers_m0 WHERE tpv_junho >= 3000
),
por_reseller AS (
  SELECT
    NIVEL_RESELLER,
    RESELLER_ID,
    COUNT(CUS_CUST_ID) AS novos_clientes,
    SUM(tpv_junho)     AS tpv_total
  FROM filtro_tpv
  GROUP BY 1,2
),
ranking AS (
  SELECT
    *,
    RANK() OVER (
      PARTITION BY NIVEL_RESELLER
      ORDER BY novos_clientes DESC, tpv_total DESC
    ) AS rank_nivel
  FROM por_reseller
)
SELECT
  t.*,
  c.CUS_CITY  AS cidade,
  c.CUS_STATE AS uf
FROM ranking t
LEFT JOIN `meli-bi-data.WHOWNER.LK_CUS_CUSTOMERS_DATA` c
  ON t.RESELLER_ID = c.CUS_CUST_ID
WHERE t.rank_nivel <= 10
ORDER BY NIVEL_RESELLER, rank_nivel
